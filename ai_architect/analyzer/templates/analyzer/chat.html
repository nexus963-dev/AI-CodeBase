def authenticate_user(credentials: LoginRequest) -> AuthResponse: # 1. Validate credentials user = db.get_user(credentials.email) if not user or not verify_hash(credentials.password, user.hash): raise AuthError("Invalid credentials") # 2. Generate tokens access_token = create_jwt(user.id, expires_in=3600) refresh_token = create_refresh_token(user.id) return AuthResponse(access=access_token, refresh=refresh_token)
